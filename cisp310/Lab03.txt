section .bss
  quo:    resb 1              ; Reserve a single byte of uninitialized data called "quo" to hold our quotient
  mod:    resb 1              ; Reserve a single byte of uninitialized data called "mod" to hold our remainder

section .data
	lbreak db 0xA      ; single byte of initialized data called "lbreak" to hold our line break character

section .text
	global _start      ; beginning of the code

_start:
  mov ax, 2          ; Place a value of 2 into the ax register
  add ax, 3          ; Add 3 to this value
  sub ax, 1          ; Subtract 1 from this value
  
  mov bx, 2          ; Place a value of 2 into the bx register
  mul bx             ; Multiply bx by ax
  
  mov cx, 3          ; Place a value of 3 into the cx register
  div cx             ; Divide ax by the cx register
  
  add ax, '0'        ; Convert the register with the quotient to ASCII
  mov [quo], ax      ; Transfer the ASCII value of the quotient to the "quo" memory address

  add dx, '0'        ; Convert the register with the remainder (modulo) to ASCII
  mov [mod], dx      ; Transfer the ASCII value of the remainder to the "mod" memory address
  
  mov edx, 1         ; message length
  mov ecx, quo       ; message to display
  mov ebx, 1         ; stdout
  mov eax, 4         ; sys_write
  int 0x80           ; system call
  
  mov eax, 4         ; The system call for write (sys_write)
  mov ebx, 1         ; stdout
  mov ecx, lbreak    ; Put the offset lbreak in ecx
  int 0x80           ; system call
  
  mov edx, 1         ; message length
  mov ecx, mod       ; message to display
  mov ebx, 1         ; stdout
  mov eax, 4         ; sys_write
  int 0x80           ; system call
  
  mov eax, 1         ; sys_exit
  mov ebx, 0         ; exit is normal
  int 0x80           ; system call
